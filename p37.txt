async function authSubmit(){
  const email=(document.getElementById("authEmail")||{}).value||"";
  const pass=(document.getElementById("authPass")||{}).value||"";
  if(!email||pass.length<6){authErr="Email и пароль от 6 символов";redrawAuth();return}
  authBusy=true;authErr="";redrawAuth();
  try{
    if(authMode==="in"){
      const{data,error}=await sb.auth.signInWithPassword({email,password:pass});
      if(error)throw error;
      await onSignedIn(data.user);
      toggleAuth(false);toast("Вход выполнен");
    }else{
      const{data,error}=await sb.auth.signUp({email,password:pass});
      if(error)throw error;
      if(!data.session){authBusy=false;authErr="Проверь почту — письмо с подтверждением отправлено";redrawAuth();return}
      await onSignedIn(data.user);
      toggleAuth(false);toast("Аккаунт создан");
    }
  }catch(e){authErr=e.message||"Не получилось";}
  authBusy=false;redrawAuth();
}

}

async function connectRealSandbox(){
  if(RS&&RS.ws&&RS.ws.readyState<=1)return;
  const{data}=await sb.auth.getSession();
  const token=data&&data.session&&data.session.access_token;
  if(!token){toast("Сессия истекла, войди заново");return}
  let ws;
  try{ws=new WebSocket(SANDBOX_URL);}catch(e){toast("Не удалось открыть соединение");return}
  RS={ws};
  ws.onopen=()=>ws.send(JSON.stringify({token}));
  ws.onmessage=(ev)=>{
    let m;try{m=JSON.parse(ev.data);}catch(e){return}
    if(m.type==="ready"){
      const st=document.getElementById("rs-status");if(st)st.textContent=`подключено · лимит ${m.minutes} мин`;
      const inp=document.getElementById("rsin");if(inp){inp.disabled=false;inp.placeholder="команда и Enter…";inp.focus();}
      rsPrint("Подключено к реальному серверу.\n");
    }else if(m.type==="out"){rsPrint(m.data);}
  };
  ws.onclose=(ev)=>{
    const st=document.getElementById("rs-status");if(st)st.textContent="отключено";
    const inp=document.getElementById("rsin");if(inp)inp.disabled=true;
    rsPrint("\n[сессия завершена: "+(ev.reason||"соединение закрыто")+"]\n");
  };
  ws.onerror=()=>toast("Не удалось подключиться к реальному серверу");
}

}
function viewComments(missionId){
  setTimeout(()=>loadCommentsAsync(missionId),0);
  const canPost=CLOUD_ENABLED&&ACCOUNT;
  return `<section class="card pad" style="margin-top:16px;text-align:left">
    <div class="lbl">💬 обсуждение миссии</div>
    ${canPost?`<div class="row" style="margin:10px 0">
        <input id="cmtinput-${missionId}" placeholder="вопрос или заметка по миссии…" maxlength="1000" style="flex:1;background:var(--panel-2);border:1px solid var(--line);border-radius:8px;color:var(--txt);padding:9px 11px;font-size:13.5px">
        <button class="btn ghost sm" data-cmtpost="${missionId}">Отправить</button></div>`
      :`<p class="muted" style="font-size:12.5px;margin:10px 0">${CLOUD_ENABLED?"Войди в аккаунт (кнопка 👤 в углу), чтобы написать комментарий.":"Комментарии появятся после настройки облака."}</p>`}
    <div id="cmts-${missionId}"><p class="muted" style="font-size:12.5px">Загрузка…</p></div>
  </section>`;
}
async function postComment(missionId){
  const el=document.getElementById("cmtinput-"+missionId);
  const body=(el?el.value:"").trim();
  if(!body)return;
  await ensureUsername();
  const{error}=await sb.from("comments").insert({mission_id:missionId,user_id:ACCOUNT.id,username:ACCOUNT.username,body});
  if(error){toast("Не отправилось: "+error.message);return}
  if(el)el.value="";
  loadCommentsAsync(missionId);
}
document.addEventListener("click",e=>{
  const t=e.target.closest("[data-cmtpost]");
  if(t)postComment(t.dataset.cmtpost);
});
document.addEventListener("keydown",e=>{
  if(e.key==="Enter"&&e.target&&e.target.id&&e.target.id.startsWith("cmtinput-")){
    e.preventDefault();postComment(e.target.id.slice("cmtinput-".length));
  }
});
