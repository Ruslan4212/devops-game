/* ================= RUNNER: миссии, боссы, повторы ================= */
let R=null,L=null;
function shuffle(a){for(let i=a.length-1;i>0;i--){const j=Math.random()*(i+1)|0;[a[i],a[j]]=[a[j],a[i]];}return a;}
const DIFF=["","разминка","крепче","сложное","очень сложное","инцидент"];
const DIFFC=["","var(--ok)","var(--amber-2)","var(--amber)","var(--warn)","var(--bad)"];
function startMission(wid,mid){
  const w=WORLDS.find(x=>x.id===wid),m=w.miss.find(x=>x.id===mid);
  // миссии с лесенкой сложности идут строго по порядку, остальные перемешиваются
  const graded=m.ch.some(c=>c.d);
  const items=graded?m.ch.slice().sort((a,b)=>(a.d||0)-(b.d||0)):shuffle(m.ch.slice());
  // возвращаемся туда, где прервались в прошлый раз
  const pos=(S.mpos||{})[mid]||{};
  const ci=Math.min(pos.ci||0,m.cards.length-1);
  R={mode:"mis",w,m,cards:m.cards,ci,items,idx:0,phase:pos.phase==="q"?"q":"card",wrong:0,ans:null,checked:false};
  if(R.phase==="q"){const it=R.items[0];if(it.t==="o")R.ord=shuffle(it.items.slice())}
  drawRunner();
}
function saveMissionPos(){
  if(!R||R.mode!=="mis")return;
  S.mpos=S.mpos||{};
  if(R.phase==="res"){delete S.mpos[R.m.id]}
  else S.mpos[R.m.id]={ci:R.ci,phase:R.phase};
  save();
}
function startBoss(wid){
  const w=WORLDS.find(x=>x.id===wid);
  R={mode:"boss",w,m:w.boss,cards:[],ci:0,items:w.boss.ch.slice(),idx:0,phase:"intro",wrong:0,hp:3,ans:null,checked:false};
  drawRunner();
}
function startRepair(){
  const d=dueList(); if(!d.length)return;
  R={mode:"rep",w:null,m:{t:"Ремонт памяти"},cards:[],ci:0,items:shuffle(d).slice(0,12),idx:0,phase:"q",wrong:0,ans:null,checked:false};
  const it=R.items[0]; if(it.t==="o")R.ord=shuffle(it.items.slice());
  drawRunner();
}
function drawRunner(){
  const v=document.getElementById("view");
  document.getElementById("tabs").innerHTML="";
  if(R.phase==="card")v.innerHTML=uiCard();
  else if(R.phase==="intro")v.innerHTML=uiIntro();
  else if(R.phase==="q")v.innerHTML=uiQ();
  else v.innerHTML=uiRes();
  const inp=v.querySelector(".term input"); if(inp)inp.focus();
  window.scrollTo({top:0});
}

function uiRes(){
  const tot=R.items.length,right=tot-R.wrong,pct=Math.round(right/tot*100),win=R.mode!=="boss"||R.hp>0;
  let extra="";
  if(R.done!==true){
    R.done=true;let xp=0;
    if(R.mode==="mis"){xp=right*12;if(!S.mis[R.m.id]){S.mis[R.m.id]=1;award("first");questTick("mission",1)}if(R.wrong===0)award("flawless")}
    else if(R.mode==="boss"&&win){xp=120;S.boss[R.w.id]=1;award(R.w.id);
      if(R.w.id==="cloudsec"){award("hired");extra=`<p style="max-width:56ch;margin:12px auto 0" class="muted">Финальный раунд пройден. Собери пет-проект из блока 10 плана, отрепетируй ответы вслух — и иди на настоящие собеседования.</p>`}}
    else if(R.mode==="rep"){xp=right*8;S.repairs+=right;if(S.repairs>=20)award("repair");questTick("repair",1)}
    R.xp=xp; if(xp){addXP(xp);checkDay()} save();
  }
  return `<div class="runner"><div class="card res">
    <div class="lbl">${R.mode==="boss"?(win?"босс повержен":"поражение"):"итог"}</div>
    <div class="big ${pct>=70?"ok":"no"}">${pct}%</div>
    <p class="muted" style="margin:6px 0 4px">${right} из ${tot} верно${R.mode==="boss"&&!win?" · жизни кончились":""}</p>
    <p class="xpfly">+${R.xp||0} XP</p>
    ${R.wrong?`<p class="muted" style="max-width:52ch;margin:12px auto 0;font-size:14px">Промахи отправлены в «Повторы» — вернутся завтра, пока не станут рефлексом.</p>`:`<p style="color:var(--ok);margin-top:10px">Чисто. Так и держи.</p>`}

      <button class="btn" data-nav="map">К карте</button>

const norm=s=>String(s||"").toLowerCase().replace(/["']/g,"").replace(/\s+/g," ").trim();
function checkAnswer(){
  const it=R.items[R.idx];
  if(it.t==="q")R.ok=R.ans===it.c;
  else if(it.t==="c"||it.t==="f"){const el=document.getElementById("cin");R.ans=el?el.value:"";
    const u=norm(R.ans);R.ok=it.ans.some(a=>norm(a)===u);}
  else if(it.t==="o")R.ok=R.ord.every((x,i)=>x===it.items[i]);
  R.checked=true;
  if(!R.ok){R.wrong++;if(R.mode==="boss")R.hp--}
  srsUpdate(it.q,R.ok,it);save();drawRunner();
}
function nextItem(){
  if(R.mode==="boss"&&R.hp<=0){R.phase="res";drawRunner();return}
  if(R.idx<R.items.length-1){R.idx++;R.ans=null;R.checked=false;R.ok=false;
    const it=R.items[R.idx];if(it.t==="o")R.ord=shuffle(it.items.slice());}
  else R.phase="res";
  drawRunner();
}
function toQuestions(){
  R.phase="q";R.idx=0;R.ans=null;R.checked=false;saveMissionPos();
  const it=R.items[0];if(it.t==="o")R.ord=shuffle(it.items.slice());
  drawRunner();
}

  L.out.push({h:`{{p|${L.ctx.user}@${L.ctx.host}:${L.ctx.cwd.replace("/home/"+L.ctx.user,"~")}$}} ${line}`});
  L.hist.push(line);L.hi=L.hist.length;

  else if(res!==""&&res!==undefined)L.out.push({h:res});
