function glossarySearch(q){
  const words=(q||"").toLowerCase().trim().split(/\s+/).filter(Boolean);
  const rows=[];
  const hay=(...parts)=>parts.join(" ").toLowerCase();
  Object.entries(GLOSSARY.commands).forEach(([cmd,info])=>{
    const cmdMatch=!words.length||words.some(w=>cmd.includes(w));
    if(!words.length||cmdMatch)rows.push({cmd,tok:cmd,d:info.d,head:true});
    Object.entries(info.sub||{}).forEach(([k,v])=>{
      const h=hay(cmd,k,v);
      if(!words.length||words.every(w=>h.includes(w)))rows.push({cmd,tok:k,d:v});
    });
    Object.entries(info.f||{}).forEach(([k,v])=>{
      const h=hay(cmd,k,v);
      if(!words.length||words.every(w=>h.includes(w)))rows.push({cmd,tok:k,d:v,flag:true});
    });
  });
  if(words.length)Object.entries(GLOSSARY.general).forEach(([k,v])=>{
    const h=hay(k,v); if(words.every(w=>h.includes(w)))rows.push({cmd:"общее",tok:k,d:v});
  });
  return rows.slice(0,150);
}
let GLOPEN=false,GLQ="";
function viewGlossary(){
  const rows=glossarySearch(GLQ);
  const groups={};
  rows.forEach(r=>{ (groups[r.cmd]=groups[r.cmd]||[]).push(r); });
  const body=Object.entries(groups).map(([cmd,items])=>`
    <div class="glogrp"><div class="mono" style="color:var(--amber-2);font-weight:700;margin:12px 0 4px">${esc(cmd)}</div>
    ${items.map(it=>`<div class="glorow"><span class="mono" style="color:${it.flag?"var(--violet)":"var(--ok)"};white-space:nowrap">${esc(it.tok)}</span><span>${esc(it.d)}</span></div>`).join("")}</div>`).join("")
    ||`<p class="muted" style="margin-top:16px">Ничего не найдено. Попробуй другое слово — например часть команды или флага.</p>`;
  return `<div class="modal-back" data-nav="gloclose"><div class="modal" onclick="event.stopPropagation()">
    <div class="row" style="margin-bottom:10px"><h2 style="font-size:19px">📖 Глоссарий команд</h2>
      <button class="btn ghost sm" data-nav="gloclose" style="margin-left:auto">Закрыть ✕</button></div>
    <input id="gloq" class="gloinput" autocomplete="off" spellcheck="false" placeholder="ищи команду, подкоманду или флаг… например: docker -p, kubectl rollout, sed -i" value="${esc(GLQ)}">
    <div class="glolist">${body}</div>
    <p class="muted" style="font-size:12px;margin-top:10px">Открывается в любой момент прохождения — если непонятен флаг в примере кода, ищи его здесь.</p>
  </div></div>`;
}
function toggleGlossary(open){
  GLOPEN=open===undefined?!GLOPEN:open;
  let host=document.getElementById("glohost");
  if(!host){host=document.createElement("div");host.id="glohost";document.body.appendChild(host);}
  host.innerHTML=GLOPEN?viewGlossary():"";
  if(GLOPEN){const el=document.getElementById("gloq");if(el){el.focus();el.setSelectionRange(el.value.length,el.value.length);}}
}

function doTabComplete(el,getCandidates){
  if(!el)return;
  const v=el.value;
  // продолжение предыдущего цикла Tab: поле не менялось с последней подстановки —
  // крутим следующий вариант из уже готового списка, не пересчитывая слово заново
  // (иначе "слово" — это уже подставленный кандидат, а не то, что печатал игрок).
  if(TABCYCLE&&TABCYCLE.id===el.id&&v===TABCYCLE.before+TABCYCLE.list[TABCYCLE.idx]){
    TABCYCLE.idx=(TABCYCLE.idx+1)%TABCYCLE.list.length;
    el.value=TABCYCLE.before+TABCYCLE.list[TABCYCLE.idx];
    if(el.setSelectionRange)el.setSelectionRange(el.value.length,el.value.length);
    return;
  }
  const sp=v.lastIndexOf(" ");
  const before=sp<0?"":v.slice(0,sp+1);
  const word=sp<0?v:v.slice(sp+1);
  if(!word){TABCYCLE=null;return}
  let list=getCandidates(word,before);
  if(!list||!list.length){TABCYCLE=null;return}
  list=[...new Set(list)].sort();
  let pick;
  if(list.length===1){
    pick=list[0];TABCYCLE=null;
  }else{
    let common=list[0];
    for(const c of list){ while(!c.startsWith(common))common=common.slice(0,-1); }
    if(common.length>word.length){pick=common;TABCYCLE=null;}
    else{pick=list[0];TABCYCLE={id:el.id,before,list,idx:0};}
  }
  el.value=before+pick;
  if(el.setSelectionRange)el.setSelectionRange(el.value.length,el.value.length);
}
function termCandidates(ctx){
  return (word,before)=>{
    if(before.trim()===""){
      return Object.keys(CMD).filter(c=>c.startsWith(word));
    }
    let dirPart="",prefix=word;
    const slash=word.lastIndexOf("/");
    if(slash>=0){dirPart=word.slice(0,slash+1);prefix=word.slice(slash+1);}
    const base=dirPart?abspath(ctx,dirPart):ctx.cwd;
    const kids=children(ctx,base);
    const names=Object.keys(kids).filter(n=>n.startsWith(prefix));
    if(!names.length)return[];
    return names.map(n=>dirPart+n+(kids[n]==="d"?"/":""));
  };
}

/* ================= РАЗБОР ИМЕННО ТВОЕЙ ОШИБКИ ================= */
function levenshtein(a,b){
  const m=a.length,n=b.length;
  if(!m)return n; if(!n)return m;
  let prev=Array.from({length:n+1},(_,j)=>j);
  for(let i=1;i<=m;i++){
    const cur=[i];
    for(let j=1;j<=n;j++)cur[j]=a[i-1]===b[j-1]?prev[j-1]:1+Math.min(prev[j-1],prev[j],cur[j-1]);
    prev=cur;
  }
  return prev[n];
}
/* сравнивает то, что реально напечатал игрок, с ближайшим правильным
   вариантом — и объясняет конкретно эту разницу, а не абстрактную теорию */
function diffAnswer(userRaw,correctList){
  const user=norm(userRaw);
  if(!user)return "Ты ничего не ввёл — начни печатать в строке терминала и нажми Enter.";
  let best=correctList[0],bestD=Infinity;
  correctList.forEach(c=>{const d=levenshtein(user,norm(c));if(d<bestD){bestD=d;best=c;}});
  const correct=norm(best);
  if(bestD>0&&bestD<=2&&Math.abs(user.length-correct.length)<=2&&user.length>3){
    return `Очень близко: ты написал «${esc(userRaw.trim())}», а нужно «${esc(best)}» — разница буквально в паре символов, присмотрись.`;
  }
  const uT=user.split(" ").filter(Boolean),cT=correct.split(" ").filter(Boolean);
  if(uT[0]&&cT[0]&&uT[0]!==cT[0]){
    return `Ты начал с «${esc(uT[0])}», а здесь нужна команда «${esc(cT[0])}» — проверь, тот ли инструмент ты вызываешь.`;
  }
  const uSet=new Set(uT),cSet=new Set(cT);
  const missing=cT.filter(t=>!uSet.has(t)),extra=uT.filter(t=>!cSet.has(t));
  if(missing.length&&!extra.length){
    return `Почти верно — в твоей команде не хватает: ${missing.map(x=>"«"+esc(x)+"»").join(", ")}.`;
  }
  if(extra.length&&!missing.length){
    return `В твоей команде лишнее: ${extra.map(x=>"«"+esc(x)+"»").join(", ")} — убери и сверься с правильным вариантом ниже.`;
  }
  if(missing.length&&extra.length){
    return `Ты написал ${extra.map(x=>"«"+esc(x)+"»").join(", ")} там, где нужно ${missing.map(x=>"«"+esc(x)+"»").join(", ")}.`;
  }
  return `Ты написал «${esc(userRaw.trim())}» — это не совпадает с ожидаемым ответом, посмотри правильный вариант ниже.`;
}

function checkTry(){
  const t=R.cards[R.ci].t;
  R.tdiag="";
  if(t.a){R.tok=R.tans===t.c; if(!R.tok)R.tdiag=diffChoice(R.tans,t);}
  else{const el=document.getElementById("tin");R.tans=el?el.value:"";
    R.tok=t.ans.some(a=>norm(a)===norm(R.tans));
    if(!R.tok)R.tdiag=diffAnswer(R.tans,t.ans);}
  R.tchecked=true;
  R.praise=pick(R.tok?PRAISE:MISSTXT);
  if(R.tok){addXP(6);if(typeof lifeReward==="function")lifeReward("drill");
    R.combo=(R.combo||0)+1;R.best=Math.max(R.best||0,R.combo);
  }else R.combo=0;
  srsUpdate(t.q,R.tok,t.a?{t:"q",q:t.q,a:t.a,c:t.c,why:t.why}:{t:"c",q:t.q,ans:t.ans,hint:t.hint,why:t.why});
  save();drawRunner();
  if(R.tok)setTimeout(()=>burst(document.querySelector(".fb.ok")),30);
}
const norm=s=>String(s||"").toLowerCase().replace(/["']/g,"").replace(/\s+/g," ").trim();
function checkAnswer(){
  const it=R.items[R.idx];
  R.diag="";
  if(it.t==="q"){R.ok=R.ans===it.c; if(!R.ok)R.diag=diffChoice(R.ans,it);}
  else if(it.t==="c"||it.t==="f"){const el=document.getElementById("cin");R.ans=el?el.value:"";
    const u=norm(R.ans);R.ok=it.ans.some(a=>norm(a)===u);
    if(!R.ok)R.diag=diffAnswer(R.ans,it.ans);}
  else if(it.t==="o")R.ok=R.ord.every((x,i)=>x===it.items[i]);
  R.checked=true;
  R.praise=pick(R.ok?PRAISE:MISSTXT);
  if(!R.ok){R.wrong++;R.combo=0;if(R.mode==="boss")R.hp--}
  else{R.combo=(R.combo||0)+1;R.best=Math.max(R.best||0,R.combo);}
  srsUpdate(it.q,R.ok,it);save();drawRunner();
  if(R.ok)setTimeout(()=>burst(document.querySelector(".fb.ok")),30);
}

}
