function fnode(v){return typeof v==="string"?{c:v,mode:"-rw-r--r--",own:"root root"}:v}
function abspath(ctx,p){
  if(!p)return ctx.cwd;
  p=p.replace(/^~/,"/home/"+ctx.user);
  let base = p.startsWith("/") ? [] : ctx.cwd.split("/").filter(Boolean);
  p.split("/").forEach(s=>{ if(!s||s===".")return; if(s==="..")base.pop(); else base.push(s); });
  return "/"+base.join("/");
}
function exists(ctx,p){ return ctx.fs[p]!==undefined || isDir(ctx,p) }
function isDir(ctx,p){ if(p==="/")return true; return Object.keys(ctx.fs).some(k=>k.startsWith(p==="/"?"/":p+"/")) }
function children(ctx,dir){
  const pre = dir==="/"?"/":dir+"/", set={};
  Object.keys(ctx.fs).forEach(k=>{ if(!k.startsWith(pre))return;
    const rest=k.slice(pre.length); if(!rest)return;
    const seg=rest.split("/")[0];
    set[seg]= rest.includes("/") ? "d" : "f"; });
  return set;
}
function pad(s,n){s=String(s);return s.length>=n?s:s+" ".repeat(n-s.length)}
function padl(s,n){s=String(s);return s.length>=n?s:" ".repeat(n-s.length)+s}
function esc(s){return String(s).replace(/&/g,"&amp;").replace(/</g,"&lt;").replace(/>/g,"&gt;")}
function fmt(s){return esc(s).replace(/\{\{(e|w|d|p|hl)\|([\s\S]*?)\}\}/g,(m,c,t)=>`<span class="${c}">${t}</span>`)}

function tokenize(s){
  const out=[];let cur="",q=null;
  for(let i=0;i<s.length;i++){const ch=s[i];
    if(q){ if(ch===q){q=null} else cur+=ch; continue }
    if(ch==='"'||ch==="'"){q=ch;continue}
    if(ch===" "){ if(cur){out.push(cur);cur=""} continue }
    cur+=ch;}
  if(cur)out.push(cur);return out;
}
function runLine(ctx,line){
  line=line.trim(); if(!line)return "";
  ctx.cmds.push(line);
  // редирект
  let redir=null,rl=line;
  const rm=line.match(/\s(>>|>)\s*(\S+)\s*$/);
  if(rm){redir={app:rm[1]===">>",path:rm[2]};rl=line.slice(0,rm.index);}
  const segs=rl.split("|").map(s=>s.trim()).filter(Boolean);
  let data=null,res="";
  for(let i=0;i<segs.length;i++){
    const tk=tokenize(segs[i]);
    if(!tk.length)continue;
    const name=tk[0],args=tk.slice(1);
    const fn=CMD[name];
    if(!fn){ return `{{e|${name}: команда не найдена}}\n{{d|введи }}help{{d| — список доступных команд}}`; }
    res=fn(ctx,args,data,name);
    data=res;
  }
  if(redir){
    const p=abspath(ctx,redir.path);
    const prev=redir.app&&ctx.fs[p]?fnode(ctx.fs[p]).c+"\n":"";
    ctx.fs[p]={c:prev+res,mode:"-rw-r--r--",own:ctx.user+" "+ctx.user};
    return "";
  }
  return res;
}

CMD.curl=(ctx,a)=>{
  ctx.flags.curl=true;
  const url=a.filter(x=>!x.startsWith("-")).pop()||"";
  const headOnly=a.some(x=>/^-.*I/.test(x)||x==="--head");
  const verbose=a.some(x=>/^-.*v/.test(x));
  const key=Object.keys(ctx.http).find(k=>url.includes(k));
  const r=key?ctx.http[key]:null;
  if(!r)return `{{e|curl: (7) Failed to connect to ${url.replace(/https?:\/\//,"").split("/")[0]}: Connection refused}}`;
  ctx.flags["curl:"+key]=true;
  const codeLine=`HTTP/1.1 ${r.code} ${r.msg||""}`;
  const hdrs=Object.entries(r.headers||{Server:"nginx/1.24.0","Content-Type":"text/html"}).map(([k,v])=>`${k}: ${v}`).join("\n");
  const pre=verbose?`{{d|*   Trying ${ctx.dns[key.replace(/https?:\/\//,"")]||"10.0.1.20"}:443...\n* TLS handshake: ${r.tls||"TLSv1.3 / OK"}\n> GET / HTTP/1.1}}\n`:"";
  const color=r.code>=500?"{{e|"+codeLine+"}}":r.code>=400?"{{w|"+codeLine+"}}":"{{p|"+codeLine+"}}";
  return pre+color+"\n"+hdrs+(headOnly?"":"\n\n"+(r.body||""));
};
