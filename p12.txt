const _git=CMD.git;
CMD.git=(ctx,a)=>{
  const G=ctx.git;
  if(a[0]==="bisect"){
    if(!G.commits)return "{{e|git bisect: в этом репозитории нечего искать}}";
    const idx=r=>{if(!r||r==="HEAD")return G.headIdx;const i=G.commits.findIndex(c=>c.h===r||c.h.startsWith(r));return i};
    const B=G.bisect=G.bisect||{};
    const step=()=>{
      if(B.lo===undefined||B.hi===undefined)return "";
      if(B.hi-B.lo<=1){B.done=true;ctx.flags.bisectFound=G.commits[B.hi].h;
        const c=G.commits[B.hi];
        return `{{hl|${c.h} is the first bad commit}}\n    ${c.m}\n{{d|(проверено ${B.steps||0} шагов; выйти из режима — git bisect reset)}}`;}
      B.cur=Math.floor((B.lo+B.hi)/2);G.headIdx=B.cur;
      const left=B.hi-B.lo-1,steps=Math.max(0,Math.round(Math.log2(left||1)));
      const c=G.commits[B.cur];
      return `Bisecting: ${left} revisions left to test after this (roughly ${steps} steps)\n[${c.h}] ${c.m}\n{{d|проверь состояние и скажи: git bisect good  или  git bisect bad}}`;
    };
    if(a[1]==="start"){B.lo=undefined;B.hi=undefined;B.done=false;B.steps=0;ctx.flags.bisectStart=true;
      return "{{d|режим поиска включён: отметь плохой и хороший коммиты}}"}
    if(a[1]==="bad"||a[1]==="good"){
      if(!ctx.flags.bisectStart)return "{{e|You need to start by \"git bisect start\"}}";
      const i=idx(a[2]);
      if(i<0)return `{{e|неизвестная ревизия: ${a[2]}}}`;
      if(a[1]==="bad")B.hi=i; else B.lo=i;
      B.steps=(B.steps||0)+1;ctx.flags.bisectSteps=B.steps;
      return step();
    }
    if(a[1]==="reset"){G.headIdx=G.commits.length-1;G.bisect={};ctx.flags.bisectReset=true;
      return "{{p|Previous HEAD position restored}}"}
    if(a[1]==="log")return (G.commits||[]).slice(0,(G.headIdx||0)+1).map(c=>`${c.h} ${c.m}`).join("\n");
    return "{{d|git bisect start | bad | good | reset}}";
  }
  if(a[0]==="show"){
    const G2=ctx.git;if(!G2.commits)return _git(ctx,a);
    const r=a[1]&&a[1]!=="HEAD"?a[1]:null;
    const c=r?G2.commits.find(x=>x.h.startsWith(r)):G2.commits[G2.headIdx];
    if(!c)return `{{e|fatal: bad object ${a[1]}}}`;
    ctx.flags["show:"+c.h]=true;
    return `commit ${c.h}\nAuthor: ${c.a||"dev@shop.ru"}\n\n    ${c.m}\n\n${c.diff||"{{d|(изменения не показаны)}}"}`;
  }
  if(a[0]==="log"&&ctx.git.commits){
    const G3=ctx.git;ctx.flags["git log"]=true;
    return G3.commits.slice().reverse().map((c,i)=>
      `{{hl|${c.h}}} ${c.m}${G3.commits.length-1-i===G3.headIdx?" {{p|(HEAD)}}":""}`).join("\n");
  }
  if(a[0]==="revert"&&ctx.git.commits){ctx.flags["revert:"+a[1]]=true;
    return `[main f0f0f0f] Revert "${(ctx.git.commits.find(c=>c.h.startsWith(a[1]))||{}).m||a[1]}"\n{{p|1 file changed}}`}
  return _git(ctx,a);
};
