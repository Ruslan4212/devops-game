function checkDay(){
  const t=today(); if(S.last===t)return;
  if(S.last){const d=Math.round((new Date(t)-new Date(S.last))/864e5);S.streak=d===1?S.streak+1:1;}else S.streak=1;
  S.last=t; if(!S.days.includes(t))S.days.push(t);

  save();

const BOX=[0,1,3,7,16,35];
function srsUpdate(key,ok,item){
  const r=S.srs[key]||{b:1};
  r.b=ok?Math.min(5,r.b+1):1;
  r.due=new Date(Date.now()+BOX[r.b]*864e5).toISOString().slice(0,10);
  r.it=item;S.srs[key]=r;
}
function dueList(){const t=today();return Object.entries(S.srs).filter(([k,v])=>v.due&&v.due<=t).map(([k,v])=>({k,...v.it}));}
const labsOf=wid=>Object.entries(LABS).filter(([id,l])=>l.w===wid).map(([id,l])=>({id,...l}));
function wProgress(w){
  const L=labsOf(w.id);
  const total=w.miss.length+L.length+1;
  const done=w.miss.filter(m=>S.mis[m.id]).length+L.filter(l=>S.labs[l.id]).length+(S.boss[w.id]?1:0);
  return {done,total,pct:Math.round(done/total*100)};
}
function wUnlocked(i){
  if(i===0)return true;
  const prev=WORLDS[i-1];
  return prev.miss.filter(m=>S.mis[m.id]).length>=prev.miss.length-1||!!S.boss[prev.id];
}
function nextThing(){
  for(let i=0;i<WORLDS.length;i++){
    const w=WORLDS[i]; if(!wUnlocked(i))break;
    const m=w.miss.find(m=>!S.mis[m.id]); if(m)return{w,m};
    const l=labsOf(w.id).find(l=>!S.labs[l.id]); if(l)return{w,lab:l};
    if(!S.boss[w.id])return{w,boss:true};
  }
  return null;
}

function ring(pct,ic){
  const c=2*Math.PI*22,off=c*(1-pct/100);
  return `<div class="ring"><svg width="52" height="52"><circle cx="26" cy="26" r="22" stroke="var(--line)" stroke-width="3" fill="none"/>
  <circle cx="26" cy="26" r="22" stroke="${pct===100?'var(--ok)':'var(--amber)'}" stroke-width="3" fill="none"
  stroke-dasharray="${c}" stroke-dashoffset="${off}" stroke-linecap="round"/></svg><b>${ic}</b></div>`;
}

}
