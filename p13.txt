const drKey=d=>d.q;
function drMastery(q){return (S.drill&&S.drill[q]&&S.drill[q].m)||0}
function drWorldMastery(w){
  const list=DR.filter(d=>d.w===w); if(!list.length)return 0;
  return list.reduce((a,d)=>a+Math.min(5,drMastery(d.q)),0)/(list.length*5);
}
function drTotalMastery(){return DR.reduce((a,d)=>a+Math.min(5,drMastery(d.q)),0)/(DR.length*5)}
function drLearned(){return DR.filter(d=>drMastery(d.q)>=3).length}
function drPick(mode,w){
  let pool=DR.slice();
  if(w)pool=pool.filter(d=>d.w===w);
  if(mode==="weak")pool.sort((a,b)=>(drMastery(a.q)-drMastery(b.q))||(Math.random()-.5));
  else if(mode==="new")pool=pool.filter(d=>drMastery(d.q)===0).concat(shuffle(pool));
  else pool=shuffle(pool);
  return pool;
}
let DRL=null;
function startDrill(mode,w){
  const pool=drPick(mode,w);
  const n=mode==="blitz"?999:12;
  DRL={mode,w,items:pool.slice(0,n),i:0,ok:0,bad:0,combo:0,best:0,xp:0,t0:Date.now(),
       limit:mode==="blitz"?60:0,done:false,checked:false,last:null};
  if(mode==="blitz")startBlitzTimer();
  drawDrill();
}
let blitzT=null;
function startBlitzTimer(){
  clearInterval(blitzT);
  blitzT=setInterval(()=>{
    if(!DRL||DRL.done){clearInterval(blitzT);return}
    const left=DRL.limit-Math.floor((Date.now()-DRL.t0)/1000);
    const el=document.getElementById("drtimer");
    if(el)el.textContent=Math.max(0,left)+"с";
    if(left<=0){clearInterval(blitzT);finishDrill()}
  },250);
}
function drCheck(val){
  const it=DRL.items[DRL.i];
  const u=norm(val);
  const ok=it.a.some(x=>norm(x)===u);
  DRL.checked=true;DRL.last={ok,val};
  const rec=S.drill[it.q]||{m:0,seen:0};
  rec.seen++;
  rec.m=ok?Math.min(5,rec.m+1):Math.max(0,rec.m-1);
  rec.at=today();
  S.drill[it.q]=rec;
  if(ok){DRL.ok++;DRL.combo++;DRL.best=Math.max(DRL.best,DRL.combo);
    const mult=1+Math.min(4,Math.floor(DRL.combo/5));
    DRL.xp+=3*mult;}
  else {DRL.bad++;DRL.combo=0;}
  save();drawDrill();
}
function drNext(){
  DRL.checked=false;DRL.last=null;DRL.i++;
  if(DRL.i>=DRL.items.length)finishDrill(); else drawDrill();
}
function finishDrill(){
  if(DRL.done)return;
  DRL.done=true;clearInterval(blitzT);
  addXP(DRL.xp);checkDay();
  S.drillRuns=(S.drillRuns||0)+1;
  S.drillOk=(S.drillOk||0)+DRL.ok;
  if(drLearned()>=40)award("drill40");
  if(drLearned()>=DR.length)award("drillall");
  if(DRL.best>=15)award("combo15");
  questTick("drill",DRL.ok);
  if(DRL.ok>=8)lifeReward("drill");
  if(DRL.mode==="blitz")questTick("blitz",1);
  save();drawDrill();
}
