function skillOf(wid){
  const w=WORLDS.find(x=>x.id===wid); if(!w)return 0;
  const labs=labsOf(wid);
  const mDone=w.miss.filter(m=>S.mis[m.id]).length/Math.max(1,w.miss.length);
  const lDone=labs.length?labs.filter(l=>S.labs[l.id]).length/labs.length:0;
  const b=S.boss[wid]?1:0;
  const dr=DR.some(d=>d.w===wid)?drWorldMastery(wid):0;
  const wm=labs.length?0.38:0.53, wl=labs.length?0.15:0, wb=0.17, wd=DR.some(d=>d.w===wid)?0.30:0;
  const norm=wm+wl+wb+wd;
  return Math.round(100*((mDone*wm+lDone*wl+b*wb+dr*wd)/norm));
}
function softSkill(){
  const iv=(S.interviews||[]).filter(x=>x.passed).length;
  const st=(S.storySeen||[]).length;
  const rp=Math.min(1,(S.repairs||0)/60);
  return Math.min(100,Math.round(iv*11+st*3+rp*25+(S.boss.cloudsec?15:0)));
}
function overall(){
  const vals=SKILLS.map(s=>skillOf(s.id));
  const avg=vals.reduce((a,b)=>a+b,0)/vals.length;
  const top=vals.slice().sort((a,b)=>b-a).slice(0,3);
  const depth=top.reduce((a,b)=>a+b,0)/3;
  return Math.round((avg*0.55+depth*0.45)*0.85+softSkill()*0.15);
}

function todayQuests(){
  const d=today();
  if(!S.quests||S.quests.d!==d){
    const seed=[...d].reduce((a,c)=>a+c.charCodeAt(0),0);
    const pool=QUESTS.slice();const pick=[];
    for(let i=0;i<3;i++)pick.push(pool.splice((seed*(i+3))%pool.length,1)[0]);
    S.quests={d,list:pick.map(q=>({id:q.id,p:0,done:false}))};save();
  }
  return S.quests.list.map(x=>({...QUESTS.find(q=>q.id===x.id),...x}));
}
function questTick(type,n){
  const qs=todayQuests();let changed=false;
  qs.forEach((q,i)=>{
    if(q.type!==type||q.done)return;
    const rec=S.quests.list[i];
    rec.p=Math.min(q.n,rec.p+(n||1));
    if(rec.p>=q.n){rec.done=true;addXP(q.xp);toast(`Задание дня: ${q.t} · +${q.xp} XP`);changed=true}
  });
  if(qs.every(q=>S.quests.list.find(x=>x.id===q.id).done)&&!S.quests.bonus){
    S.quests.bonus=true;addXP(50);award("dailyall");toast("Все задания дня закрыты · +50 XP");
  }
  if(changed)save();
}

let IV=null;
function startInterview(id){
  const j=JOBS.find(x=>x.id===id);
  const reqOk=Object.entries(j.req).every(([k,v])=>skillOf(k)>=v-12);
  const pool=[];
  j.w.forEach(wid=>{const w=WORLDS.find(x=>x.id===wid);
    w.miss.forEach(m=>m.ch.forEach(c=>{if(c.t==="q")pool.push({...c,w:wid})}));
    w.boss.ch.forEach(c=>{if(c.t==="q")pool.push({...c,w:wid})});});
  const soft=shuffle(SOFTQ.slice()).slice(0,gradeIdx(j.g)>=3?3:2).map(c=>({...c,w:"soft"}));
  const tech=shuffle(pool).slice(0,Math.max(2,j.qn-soft.length));
  IV={j,phase:reqOk?"brief":"screen",items:shuffle(tech.concat(soft)),i:0,ok:0,ans:null,checked:false,reqOk};
  drawIV();
}
function ivFeedback(){
  const j=IV.j;
  return Object.entries(j.req).map(([k,v])=>{
    const cur=skillOf(k),s=SKILLS.find(x=>x.id===k);
    return {n:s.n,cur,need:v,gap:v-cur,w:k};
  }).sort((a,b)=>b.gap-a.gap);
}

function ivFinish(){
  const j=IV.j;
  const bonus=(typeof interviewBonus==="function"&&S.life&&S.life.created)?interviewBonus():0;
  IV.bonus=bonus;
  const score=IV.ok/IV.items.length+bonus/100;
  const passed=score>=j.pass;
  if(passed&&j.lab&&!S.labs[j.lab]){IV.phase="task";drawIV();return}
  IV.passed=passed;IV.phase="res";
  S.interviews=(S.interviews||[]).concat([{id:j.id,passed,score:IV.ok,total:IV.items.length,d:today()}]);
  addXP(passed?90:35);questTick("interview",1);checkDay();if(typeof lifeReward==="function")lifeReward("interview");
  if(passed)award("firstoffer");
  if((S.interviews||[]).length>=5)award("iv5");
  save();drawIV();
}
function acceptOffer(id){
  const j=JOBS.find(x=>x.id===id);
  S.job={id,since:today()};
  S.rep=(S.rep||0)+10;
  addXP(150);
  if(j.g==="middle")award("middle");
  save();toast(`Ты принят в «${j.n}»`);go("career");
}
