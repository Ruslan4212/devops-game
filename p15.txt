function viewCareer(){
  const r=rank(),o=overall(),j=S.job?JOBS.find(x=>x.id===S.job.id):null;
  const qs=todayQuests();
  const myIdx=j?gradeIdx(j.g):-1;
  const cards=JOBS.map(job=>{
    const reqs=Object.entries(job.req).map(([k,v])=>({k,v,cur:skillOf(k)}));
    const ready=reqs.every(x=>x.cur>=x.v);
    const close=reqs.every(x=>x.cur>=x.v-12);
    const tooLow=gradeIdx(job.g)>myIdx+2&&!close;
    const attempts=(S.interviews||[]).filter(x=>x.id===job.id);
    const got=attempts.some(x=>x.passed);
    const cur=j&&j.id===job.id;
    return `<div class="card pad" style="border-color:${cur?"rgba(63,191,143,.5)":ready?"rgba(240,135,63,.5)":"var(--line)"};opacity:${tooLow?.5:1}">
      <div class="row" style="align-items:flex-start">
        <div style="flex:1;min-width:220px">
          <div class="row" style="gap:8px"><span class="chip ${ready?"lab":""}">${gradeName(job.g)}</span>
            <span class="chip">${job.pay}</span>${cur?`<span class="chip done">твоё место работы</span>`:got?`<span class="chip done">был оффер</span>`:""}</div>
          <h3 style="font-size:19px;margin:8px 0 2px">${job.n}</h3>
          <div class="lbl">${job.tag}</div>
          <p class="muted" style="margin:8px 0 0;font-size:13.5px">${job.desc}</p>
        </div>
        <div style="min-width:210px">
          <div class="lbl" style="margin-bottom:6px">требования</div>
          ${reqs.map(x=>{const s=SKILLS.find(y=>y.id===x.k);
            return `<div class="row" style="justify-content:space-between;font-size:12.5px;margin-bottom:3px">

      <div class="lbl">карьера</div>

document.addEventListener("click",e=>{
  const t=e.target.closest("[data-drill],[data-career],[data-story]");
  if(!t)return;
  const d=t.dataset.drill,c=t.dataset.career,s=t.dataset.story;
  if(d){
    if(d==="check"){const i=document.getElementById("drin");drCheck(i?i.value:"");return}
    if(d==="next"){drNext();return}
    if(d==="stop"){finishDrill();return}
    if(d==="again"){startDrill(DRL.mode,DRL.w);return}
    if(d.startsWith("world:")){startDrill("world",d.split(":")[1]);return}
    startDrill(d);return;
  }
  if(s){
    const [id,i]=s.split(":");const sc=STORY.find(x=>x.id===id);
    S.storySeen=(S.storySeen||[]).concat([id]);
    S.storyPick=S.storyPick||{};S.storyPick[id]=sc.ch[+i].e;
    addXP(40);checkDay();save();
    toast(sc.ch[+i].e);
    render();return;
  }
  if(c){
    if(c.startsWith("apply:")){startInterview(c.split(":")[1]);return}
    if(c.startsWith("retry:")){startInterview(c.split(":")[1]);return}
    if(c==="ivgo"){IV.phase="q";IV.i=0;IV.ans=null;IV.checked=false;drawIV();return}
    if(c.startsWith("ivopt:")){if(IV.checked)return;IV.ans=+c.split(":")[1];drawIV();return}
    if(c==="ivcheck"){IV.checked=true;if(IV.ans===IV.items[IV.i].c)IV.ok++;
      srsUpdate(IV.items[IV.i].q,IV.ans===IV.items[IV.i].c,IV.items[IV.i]);save();drawIV();return}
    if(c==="ivnext"){if(IV.i<IV.items.length-1){IV.i++;IV.ans=null;IV.checked=false;drawIV()}else ivFinish();return}
    if(c==="ivlab"){const id=IV.j.lab;IV.pendingLab=id;startLab(id);return}
    if(c.startsWith("accept:")){acceptOffer(c.split(":")[1]);return}
    if(c.startsWith("goworld:")){S.openWorld=c.split(":")[1];go("map");return}
  }
});
document.addEventListener("keydown",e=>{
  const el=document.getElementById("drin");
  if(DRL&&!DRL.done&&el&&document.activeElement===el&&e.key==="Enter"){
    e.preventDefault();DRL.checked?drNext():drCheck(el.value);return;
  }
  if(IV&&IV.phase==="q"){
    if(/^[1-9]$/.test(e.key)&&!IV.checked){const i=+e.key-1;if(i<IV.items[IV.i].a.length){IV.ans=i;drawIV()}}
    else if(e.key==="Enter"){e.preventDefault();
      if(IV.checked){if(IV.i<IV.items.length-1){IV.i++;IV.ans=null;IV.checked=false;drawIV()}else ivFinish()}
      else if(IV.ans!==null){IV.checked=true;if(IV.ans===IV.items[IV.i].c)IV.ok++;save();drawIV()}}
  }
});
