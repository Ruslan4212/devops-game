async function downloadCertificate(){
  const r=rank();
  if(r.g!=="middle"){toast("Сертификат откроется по достижении ранга Middle DevOps");return}
  toast("Готовлю сертификат…");
  const cv=await buildCertificateCanvas({
    username:(typeof ACCOUNT!=="undefined"&&ACCOUNT&&ACCOUNT.username)?ACCOUNT.username:"Player",
    xp:S.xp,missions:Object.keys(S.mis).length,labs:Object.keys(S.labs).length,
    bosses:Object.keys(S.boss).length,badges:S.badges.length,
    date:new Date().toLocaleDateString("ru-RU",{year:"numeric",month:"long",day:"numeric"})
  });
  cv.toBlob(b=>{
    const url=URL.createObjectURL(b);
    const a=document.createElement("a");a.href=url;a.download="pipeline-middle-devops-certificate.png";
    document.body.appendChild(a);a.click();a.remove();
    setTimeout(()=>URL.revokeObjectURL(url),4000);
    toast("Сертификат скачан ✓");
  },"image/png");
}
