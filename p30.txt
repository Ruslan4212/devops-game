function statBar(label,val,color,warn){
  return `<div style="flex:1;min-width:120px">
    <div class="row" style="justify-content:space-between;margin-bottom:3px">
      <span class="lbl">${label}</span><span class="mono" style="color:${warn?"var(--bad)":color}">${Math.round(val)}</span></div>
    <div class="xpbar" style="width:100%"><i style="width:${Math.max(0,Math.min(100,val))}%;background:${warn?"var(--bad)":color}"></i></div></div>`;
}

}

/* ---------- сводка за прошедшие дни ---------- */
function lifeReport(r){
  if(!r||(!r.days))return;
  const l=LF();
  let msg=`Прошло ${r.days} ${r.days===1?"день":"дн."}`;
  if(r.skipped)msg+=` (тебя не было дольше, но списываем не больше трёх дней)`;
  const parts=[];
  if(r.earned)parts.push("заработано "+money(r.earned));
  if(r.spent)parts.push("потрачено "+money(r.spent));
  if(parts.length)msg+=": "+parts.join(", ");
  toast(msg);
  if(r.events.length)setTimeout(()=>toast(r.events[0]),3000);
}
