/* ---------- экономика ---------- */
function LF(){S.life=S.life||defLife();return S.life}
function dailySalary(){
  if(!S.job)return 0;
  const j=JOBS.find(x=>x.id===S.job.id); if(!j)return 0;
  return Math.round(parseInt(j.pay.replace(/\D/g,""),10)/30);
}
function dailyCosts(){
  const l=LF();let c=0;
  if(l.home){const h=HOMES.find(x=>x.id===l.home);
    if(h&&!l.own.includes("home:"+h.id))c+=Math.round(h.rent/30);
    c+=UTILITY_DAY;}
  if(l.car){const car=CARS.find(x=>x.id===l.car);if(car)c+=Math.round(car.up/30)}
  return c;
}
function techXPBonus(){const l=LF();return l.tech.reduce((a,id)=>a+((TECH.find(t=>t.id===id)||{}).xp||0),0)
  +(l.car?((CARS.find(c=>c.id===l.car)||{}).xp||0):0)}
function moodXPMod(){const m=LF().mood;return m>=80?10:m>=40?0:m>=20?-10:-25}
function xpMultiplier(){return 1+(techXPBonus()+moodXPMod())/100}
function interviewBonus(){
  const l=LF();let b=0;
  ["top","shoes"].forEach(sl=>{const it=CLOTHES.find(c=>c.id===l.wear[sl]);if(it)b+=it.iv});
  if(l.mood>=70)b+=3; if(l.mood<25)b-=5;
  if(l.hunger<20)b-=5;
  return b;
}
function lifeAdd(txt,sum){
  const l=LF();
  l.log.unshift({d:today(),t:txt,s:sum||0});
  if(l.log.length>40)l.log.length=40;
}
function earn(sum,why){const l=LF();l.money+=sum;l.totalEarned+=sum;if(why)lifeAdd(why,sum);save()}
function spend(sum,why){const l=LF();if(l.money<sum)return false;l.money-=sum;l.totalSpent+=sum;lifeAdd(why,-sum);save();return true}

/* ---------- суточный тик ---------- */
function lifeTick(){
  const l=LF(); if(!l.created)return null;
  const t=today();
  if(!l.day){l.day=t;save();return null}
  if(l.day===t)return null;
  let days=Math.round((new Date(t)-new Date(l.day))/864e5);
  const skipped=days>3?days-3:0;
  days=Math.min(days,3);            // за долгое отсутствие штрафуем максимум за три дня
  const r={days,skipped,earned:0,spent:0,events:[]};
  for(let i=0;i<days;i++){
    if(l.dead)break;
    l.daysLived++;
    const sal=dailySalary();
    if(sal){l.money+=sal;l.totalEarned+=sal;r.earned+=sal}
    const cost=dailyCosts();
    if(cost){
      if(l.money>=cost){l.money-=cost;l.totalSpent+=cost;r.spent+=cost}
      else{l.money=0;l.mood-=10;r.events.push("Не хватило денег на счета: настроение упало")}
    }
    l.hunger-=30;
    if(l.hunger<0){l.health+=Math.round(l.hunger/2);l.hunger=0;r.events.push("Голодный день: здоровье падает")}
    const h=l.home?HOMES.find(x=>x.id===l.home):null;
    l.mood+=h?(h.mood-3):STREET_MOOD;
    if(!h){l.health+=STREET_HEALTH;r.events.push("Ночёвка без жилья бьёт по здоровью")}
    if(l.hunger>60&&l.health<100)l.health+=4;
    l.mood=Math.max(0,Math.min(100,l.mood));
    l.hunger=Math.max(0,Math.min(100,l.hunger));
    l.health=Math.max(0,Math.min(100,l.health));
    if(l.health<=0){lifeDeath();r.events.push("Организм не выдержал");break}
  }
  l.day=t;
  if(r.earned)lifeAdd("Зарплата за "+days+" дн.",r.earned);
  if(r.spent)lifeAdd("Жильё, коммуналка, авто",-r.spent);
  save();
  return r;
}
function lifeDeath(){
  const l=LF();
  l.dead=true;l.deaths++;
  l.money=0;l.car=null;
  if(l.home&&!l.own.includes("home:"+l.home))l.home=null;   // съёмное теряется, купленное остаётся
  S.job=null;
  lifeAdd("Истощение. Работа потеряна, накопления ушли на лечение",0);
  save();
}
function lifeRevive(){

  lifeAdd("Выписан из больницы. Начинаем заново",0);

}
function eat(id){
  const l=LF(),f=FOOD.find(x=>x.id===id); if(!f)return "нет такого";
  if(l.money<f.p)return "не хватает денег";
  spend(f.p,"Еда: "+f.n);
  l.hunger=Math.min(100,l.hunger+f.h);
  l.mood=Math.max(0,Math.min(100,l.mood+f.m));
  if(l.hunger>50&&l.health<100)l.health=Math.min(100,l.health+3);
  save();return null;
}
